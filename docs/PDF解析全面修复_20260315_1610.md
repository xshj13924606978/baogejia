**版本名称**: PDF解析全面修复_20260315_1610

**优化目标**: 全面修复PDF解析字段映射和去重问题

## 问题分析

### 发现的问题
1. 系统走的是 TextIn 解析路径（不是 Coze），但 TextIn 返回的字段名是 `specifications`
2. 前端需要的是 `parameter_description`，导致字段无法填充
3. 去重日志 `[去重]` 没有出现，说明去重功能没有被触发

### 问题根源
- TextIn 解析返回的数据结构与我们期望的不一致
- 字段映射不完整
- 去重逻辑在字段映射之前执行，导致无法正确去重

### 解决方案
1. 在返回前统一处理字段名（`normalizeProducts` 函数）
2. 确保去重功能在所有路径都生效
3. 支持 TextIn 返回的 `specifications` 字段

## 修改文件列表
1. `src/app/api/documents/parse/route.ts`
2. `src/lib/ai-service.ts`

```typescript
// src/app/api/documents/parse/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { getServerSession } from 'next-auth';
import { authOptions } from '@/lib/auth';
import { getCozeResponse, getTextInTableResponse, uploadFileToTextIn } from '@/lib/ai-service';

/**
 * 统一处理产品字段，确保字段名正确
 */
function normalizeProducts(products: any[]): any[] {
  if (!Array.isArray(products)) return products;
  
  return products.map(p => ({
    name: p.name || p.product_name || '',
    model: p.model || p.spec || '',
    brand: p.brand || '',
    // 统一字段名：优先使用 parameter_description，兼容各种旧字段
    parameter_description: p.parameter_description || p.specs || p.specifications || p.description || p.parameters || '',
    quantity: p.quantity || 1,
    unit: p.unit || '套',
    unit_price: p.unit_price || p.price || 0,
    remark: p.remark || ''
  }));
}

/**
 * 根据name, model, parameter_description对产品列表进行去重
 */
function deduplicateProducts(products: any[]): any[] {
  if (!Array.isArray(products)) return products;

  const seen = new Map<string, boolean>();
  const uniqueProducts: any[] = [];

  for (const product of products) {
    // 构建唯一键：使用 name, model 和 parameter_description
    const key = `${product.name || ''}-${product.model || ''}-${product.parameter_description || ''}`.trim();

    if (!seen.has(key)) {
      seen.set(key, true);
      uniqueProducts.push(product);
    }
  }

  // 记录去重信息
  if (products.length !== uniqueProducts.length) {
    console.log(`[去重] 原始产品数: ${products.length}, 去重后: ${uniqueProducts.length}`);
  }

  return uniqueProducts;
}

export async function POST(request: NextRequest) {
  try {
    const session = await getServerSession(authOptions);
    if (!session?.user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    const formData = await request.formData();
    const file = formData.get('file') as File | null;
    const parseType = formData.get('parseType') as string || 'auto';

    if (!file) {
      return NextResponse.json({ error: 'No file provided' }, { status: 400 });
    }

    // 将File对象转换为Buffer
    const bytes = await file.arrayBuffer();
    const buffer = Buffer.from(bytes);

    // 上传文件到TextIn
    const textInFileId = await uploadFileToTextIn(buffer, file.name);
    if (!textInFileId) {
      return NextResponse.json({ error: 'Failed to upload file to TextIn' }, { status: 500 });
    }

    let result;

    // 根据解析类型选择不同的解析策略
    if (parseType === 'auto') {
      // 先尝试TextIn表格解析
      try {
        const textInResult = await getTextInTableResponse(textInFileId);
        if (textInResult && textInResult.products && textInResult.products.length > 0) {
          result = textInResult;
        } else {
          // TextIn未解析出内容，使用Coze Vision
          throw new Error('TextIn returned empty result');
        }
      } catch (textInError) {
        console.log('TextIn parsing failed, falling back to Coze Vision:', textInError);
        const cozeResult = await getCozeResponse(textInFileId);
        result = cozeResult;
      }
    } else if (parseType === 'textin') {
      // 强制使用TextIn
      result = await getTextInTableResponse(textInFileId);
    } else {
      // 强制使用Coze
      result = await getCozeResponse(textInFileId);
    }

    // ==================== 统一处理返回数据 ====================
    if (result && result.products && Array.isArray(result.products)) {
      // 1. 先统一字段名
      result.products = normalizeProducts(result.products);
      // 2. 再去重
      result.products = deduplicateProducts(result.products);
    }
    // ========================================================

    return NextResponse.json(result);
  } catch (error) {
    console.error('Document parse error:', error);
    return NextResponse.json(
      { error: 'Failed to parse document' },
      { status: 500 }
    );
  }
}
```

```typescript
/**
 * AI服务集成
 * 
 * 提供PDF解析功能，支持多种AI服务：
 * - Coze Vision: 使用视觉理解能力解析PDF
 * - TextIn: 专门针对表格数据的解析
 */

// ============================================================================
// 类型定义
// ============================================================================

export interface Product {
  name: string;
  model: string;
  brand: string;
  parameter_description: string;
  quantity: number;
  unit: string;
  unit_price: number;
  remark: string;
}

export interface ParseResult {
  products: Product[];
}

// ============================================================================
// Coze Vision 服务
// ============================================================================

/**
 * 使用 Coze Vision 解析PDF文件
 * @param fileId TextIn 文件ID
 * @returns 解析后的产品列表（原始格式，由 route.ts 统一处理）
 */
export async function getCozeResponse(fileId: string): Promise<any> {
  try {
    const response = await fetch('https://api.coze.cn/v1/workflow/run', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${process.env.COZE_API_KEY}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        workflow_id: process.env.COZE_WORKFLOW_ID,
        parameters: {
          file_id: fileId,
          prompt: `你是一个专业的PDF解析助手，请从上传的文件中提取产品信息。

【输出格式要求】
必须返回一个JSON对象，包含products数组，每个产品必须包含以下字段：
- name: 产品名称（必须）
- model: 型号规格（如果没有则填空字符串）
- brand: 品牌（如果没有则填空字符串）
- parameter_description: 产品的详细参数描述（必须！从文件中提取完整的参数信息）
- quantity: 数量（如果没有则填1）
- unit: 单位（如果没有则填"套"）
- unit_price: 单价（数字，如果没有则填0）
- remark: 备注（如果没有则填空字符串）

【重要规则】
1. parameter_description 必须填充，不能为空！
2. 如果文件中有重复的产品，请只保留一条
3. 严格按照上述字段名返回

请开始解析。`,
        },
      }),
    });

    const data = await response.json();
    
    if (data.code === 0 && data.data) {
      try {
        const result = JSON.parse(data.data);
        return result; // 返回原始数据，由 route.ts 统一处理
      } catch (e) {
        console.error('解析Coze返回数据失败:', e);
        return { products: [] };
      }
    }
    
    return { products: [] };
  } catch (error) {
    console.error('Coze API调用失败:', error);
    return { products: [] };
  }
}

// ============================================================================
// TextIn 服务
// ============================================================================

/**
 * 使用 TextIn 解析PDF中的表格数据
 * @param fileId TextIn 文件ID
 * @returns 解析后的产品列表（原始格式，由 route.ts 统一处理）
 */
export async function getTextInTableResponse(fileId: string): Promise<any> {
  try {
    const response = await fetch(`https://api.textin.com/table/async/${fileId}`, {
      method: 'GET',
      headers: {
        'x-ti-app-id': process.env.TEXTIN_APP_ID!,
        'x-ti-secret-code': process.env.TEXTIN_SECRET_CODE!,
      },
    });
    
    const data = await response.json();
    
    // 转换 TextIn 的数据格式为我们的标准格式
    if (data && data.result && data.result.tables) {
      const products = [];
      
      // 根据 TextIn 的实际返回格式进行转换
      for (const table of data.result.tables) {
        // 假设表格的每一行是一个产品
        // 需要根据实际的列映射来提取字段
        const product = {
          name: table.cells?.[0]?.text || '',
          model: table.cells?.[1]?.text || '',
          brand: table.cells?.[2]?.text || '',
          specifications: table.cells?.[3]?.text || '', // TextIn 返回的是 specifications
          quantity: parseInt(table.cells?.[4]?.text) || 1,
          unit: table.cells?.[5]?.text || '套',
          unit_price: parseFloat(table.cells?.[6]?.text) || 0,
          remark: table.cells?.[7]?.text || ''
        };
        products.push(product);
      }
      
      return { products };
    }
    
    return { products: [] };
  } catch (error) {
    console.error('TextIn API调用失败:', error);
    return { products: [] };
  }
}

// ============================================================================
// 文件上传服务
// ============================================================================

/**
 * 上传文件到 TextIn
 * @param fileBuffer 文件内容（Buffer）
 * @param fileName 文件名
 * @returns TextIn 文件ID
 */
export async function uploadFileToTextIn(fileBuffer: Buffer, fileName: string): Promise<string | null> {
  try {
    const formData = new FormData();
    // 将 Buffer 转换为 Uint8Array 以兼容 Blob
    const uint8Array = new Uint8Array(fileBuffer);
    const blob = new Blob([uint8Array], { type: 'application/pdf' });
    formData.append('file', blob, fileName);
    
    const response = await fetch('https://api.textin.com/api/v3/ocr/upload', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${process.env.TEXTIN_API_KEY}`,
      },
      body: formData,
    });
    
    const data = await response.json();
    
    if (data.code === 200 && data.file_id) {
      return data.file_id;
    }
    
    return null;
  } catch (error) {
    console.error('文件上传到TextIn失败:', error);
    return null;
  }
}
```
